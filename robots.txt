# Lo que pide sesión lleva además <meta name="robots" content="noindex"> en la
# propia página: esto es el primer filtro, no el único. cursos/protegido/ son
# los fragmentos con las lecciones completas, que no son páginas: los pide el
# navegador desde la portada del curso cuando hay sesión.

User-agent: *
Allow: /

Disallow: /cursos/protegido/
# El material de estudio de cada lección (cuadernillos y presentaciones) es de
# uso docente: no tiene sentido que salga en una búsqueda. Como siempre acá,
# esto no es la puerta: la puerta la pone worker.js, que solo le sirve estas dos
# carpetas a una cuenta de la Academia con el acceso vigente.
Disallow: /cursos/recursos/
Disallow: /cursos/academia/
# El entrenamiento pide sesión, salvo el diagnóstico de nivel: ese se hace sin
# cuenta y es la puerta de entrada al sitio, así que tiene que poder indexarse.
# El Allow va ANTES y es más específico, que es como se desempata: sin esta
# línea, quitarle el noindex a la página no habría servido de nada — Google ni
# siquiera llegaría a leerlo.
Allow: /entreno/diagnostico.html
Disallow: /entreno/
Disallow: /admin.html
Disallow: /arbitraje.html
Disallow: /informes.html
Disallow: /sesion.html
Disallow: /clases.html
Disallow: /configuracion.html
Disallow: /login.html
